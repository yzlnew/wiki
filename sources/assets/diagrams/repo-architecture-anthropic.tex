\newcommand{\anthropicLabel}[4]{%
  \def\tmp{#4}%
  \ifx\tmp\empty
    {\sffamily\bfseries\small\textcolor{#1}{#3}}%
  \else
    \begin{tabular}{@{}c@{}}
      {\sffamily\bfseries\small\textcolor{#1}{#3}}\\[-1pt]
      {\sffamily\fontsize{8}{9}\selectfont\textcolor{#2}{#4}}
    \end{tabular}%
  \fi
}
